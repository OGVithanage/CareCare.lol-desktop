cmake_minimum_required(VERSION 3.24)
project(CareCareWfp LANGUAGES CXX)
add_library(CareCareWfp SHARED wfp.cpp)
target_compile_features(CareCareWfp PRIVATE cxx_std_17)
target_compile_definitions(CareCareWfp PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CareCareWfp PRIVATE fwpuclnt ws2_32 dnsapi rpcrt4)
if(MSVC)
  target_compile_options(CareCareWfp PRIVATE /W4 /WX /EHsc)
  set_property(TARGET CareCareWfp PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
